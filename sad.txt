"""
Experiment 3: CRC (Cyclic Redundancy Check) error detection for ASCII text
Run:  python exp3_crc.py
Idea: Sender divides data by a generator using XOR (modulo-2) division and
      appends the remainder (CRC). Receiver divides again: remainder 0 = OK.
"""

def ascii_to_binary(text):
    # Each character -> 8-bit binary string. 'A' (65) -> '01000001'
    return ''.join(format(ord(ch), '08b') for ch in text)


def mod2_divide(dividend, generator):
    """Modulo-2 (XOR) division. Returns the remainder (n-1 bits)."""
    n = len(generator)
    rem = list(dividend)                      # list so we can change bits
    for i in range(len(dividend) - n + 1):
        if rem[i] == '1':                     # divide only when leading bit is 1
            for j in range(n):                # XOR generator with current bits
                rem[i + j] = '0' if rem[i + j] == generator[j] else '1'
    return ''.join(rem[-(n - 1):])            # last (n-1) bits = remainder


def main():
    message = input("Enter ASCII message: ")
    generator = input("Enter generator polynomial bits (e.g. 1101): ").strip()
    n = len(generator)

    # ---------- SENDER SIDE ----------
    data = ascii_to_binary(message)
    appended = data + '0' * (n - 1)           # add (n-1) zeros before dividing
    crc = mod2_divide(appended, generator)    # CRC checksum
    frame = data + crc                        # frame actually transmitted
    print("\nData bits         :", data)
    print("CRC (checksum)    :", crc)
    print("Transmitted frame :", frame)

    # ---------- CHANNEL (optional error) ----------
    received = frame
    if input("\nSimulate a transmission error? (y/n): ").lower() == 'y':
        pos = int(input(f"Bit position to flip (0-{len(frame) - 1}): "))
        flipped = '0' if received[pos] == '1' else '1'
        received = received[:pos] + flipped + received[pos + 1:]
        print("Received (with error):", received)

    # ---------- RECEIVER SIDE ----------
    remainder = mod2_divide(received, generator)
    print("\nReceiver remainder:", remainder)
    if '1' in remainder:                      # any 1 = non-zero remainder
        print("Result: Error detected! Asking sender to retransmit...")
        # CRC only DETECTS errors; correction = retransmission (ARQ)
        print("Retransmitted frame:", frame)
        print("Receiver remainder after retransmission:",
              mod2_divide(frame, generator), "-> data accepted")
    else:
        print("Result: No error detected. Data accepted.")


if __name__ == "__main__":
    main()

"""
Experiment 4a: Go-Back-N ARQ simulation
Run:  python exp4_go_back_n.py
Idea: Sender sends a window of frames. If one frame is lost, receiver discards
      all later frames, so sender goes back and resends from the lost frame.
"""

total = int(input("Enter total number of frames: "))
window = int(input("Enter window size (N): "))
lost = int(input("Frame to simulate as lost (0-indexed, -1 for none): "))

base = 0              # first frame of the current window
retransmissions = 0   # count of frames sent again

while base < total:
    end = min(base + window, total)           # window = frames base .. end-1
    print(f"\nSending window: frames {base} to {end - 1}")

    if base <= lost < end:                    # loss happens inside this window
        for i in range(base, end):
            if i < lost:
                print(f"  Frame {i} -> Acknowledged")
            elif i == lost:
                print(f"  Frame {i} -> LOST in transit")
            else:
                print(f"  Frame {i} -> Discarded by receiver (out of order)")
        print(f"  Timeout! Go-Back-N: resending frames {lost} to {end - 1}")
        retransmissions += end - lost         # all frames from lost onward
        base = lost                           # GO BACK to the lost frame
        lost = -1                             # assume it reaches next time
    else:
        for i in range(base, end):
            print(f"  Frame {i} -> Acknowledged")
        base = end                            # slide window forward

print("\nTotal frames retransmitted (Go-Back-N):", retransmissions)

"""
Experiment 4b: Selective Repeat ARQ simulation
Run:  python exp4_selective_repeat.py
Idea: Receiver keeps correctly received out-of-order frames. Only the lost
      frame is retransmitted.
"""

total = int(input("Enter total number of frames: "))
window = int(input("Enter window size (N): "))
lost = int(input("Frame to simulate as lost (0-indexed, -1 for none): "))

acked = [False] * total   # acked[i] = True when frame i is received
retransmitted = []        # list of frames that had to be resent
base = 0

while base < total:
    end = min(base + window, total)
    print(f"\nSending window: frames {base} to {end - 1}")

    for i in range(base, end):
        if i == lost:
            print(f"  Frame {i} -> LOST in transit")
        else:
            acked[i] = True                   # receiver buffers it & ACKs it
            print(f"  Frame {i} -> Received & individually ACKed")

    if base <= lost < end:                    # timeout only for the lost frame
        print(f"  Timeout for frame {lost} only. Resending frame {lost}")
        acked[lost] = True
        retransmitted.append(lost)
        lost = -1

    while base < total and acked[base]:       # slide window past ACKed frames
        base += 1

print("\nTotal frames retransmitted (Selective Repeat):", len(retransmitted))

"""
Experiment 5c: TCP calculator SERVER   (start this FIRST)
Run:  python exp5c_calc_server.py
Protocol: client sends text like "12 + 7", server replies with the answer.
"""
import socket

server = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
server.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
server.bind(("0.0.0.0", 9003))
server.listen(5)
print("Calculator server listening on port 9003...")

conn, addr = server.accept()
expr = conn.recv(256).decode().strip()       # e.g. "12 + 7"
print("Received:", expr)

try:
    a, op, b = expr.split()                  # split into 3 parts
    a, b = float(a), float(b)
    if op == '+':   result = a + b
    elif op == '-': result = a - b
    elif op == '*': result = a * b
    elif op == '/': result = a / b           # b == 0 raises ZeroDivisionError
    else: raise ValueError("bad operator")
    reply = f"{result:g}"                    # 19.0 -> '19'
except ZeroDivisionError:
    reply = "Error: division by zero"
except Exception:
    reply = "Error: invalid expression"

conn.send(reply.encode())
conn.close()
server.close()

"""
Experiment 7b: TCP Multiuser chat SERVER (broadcasts to everyone)
Run:  python exp7b_multiuser_server.py
"""
import socket
import threading

clients = []                       # all connected client sockets
lock = threading.Lock()            # protects 'clients' from thread clashes


def broadcast(message, sender):
    """Send message to every client except the sender."""
    with lock:
        for c in clients:
            if c is not sender:
                try:
                    c.send(message.encode())
                except OSError:
                    pass           # that client probably left


def handle_client(conn, addr):
    """One thread per client: read its messages and broadcast them."""
    name = f"{addr[0]}:{addr[1]}"
    while True:
        try:
            data = conn.recv(1024)
        except OSError:
            break
        if not data:               # client disconnected
            break
        msg = f"[{name}]: {data.decode()}"
        print(msg)                 # server console logs everything
        broadcast(msg, conn)
    with lock:
        if conn in clients:
            clients.remove(conn)
    conn.close()
    print(f"{name} disconnected.")


server = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
server.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
server.bind(("0.0.0.0", 9201))
server.listen(10)
print("Multiuser chat server listening on port 9201...")

while True:
    conn, addr = server.accept()
    with lock:
        clients.append(conn)
    print(f"{addr[0]}:{addr[1]} connected.")
    threading.Thread(target=handle_client, args=(conn, addr), daemon=True).start()

Experiment No. 3

Aim: To write a program that performs error detection (and demonstrates correction/retransmission) for ASCII-coded data using the Cyclic Redundancy Check (CRC) technique.

Objectives:
• Understand how CRC generates a checksum using polynomial (modulo-2) division.
• Implement CRC generation at the sender and verification at the receiver.
• Simulate a transmission error and observe CRC's error-detection capability.

Theory:
Cyclic Redundancy Check (CRC) is an error-detecting code commonly used in digital networks and storage devices. The sender treats the data bits as a polynomial and divides it (using modulo-2/XOR arithmetic) by a fixed generator polynomial; the remainder is appended to the data as the CRC/FCS (Frame Check Sequence) before transmission.

At the receiver, the same division is performed on the received bit-stream (data + CRC) using the same generator polynomial. If the remainder is zero, the data is assumed to be error-free; a non-zero remainder indicates that the data was corrupted during transmission.

CRC is capable of detecting all single-bit errors, all double-bit errors (for a suitably chosen generator), and most burst errors, making it far more reliable than a simple parity check, though it detects errors rather than correcting them directly. Correction is typically achieved by requesting retransmission (ARQ).


Experiment No. 4

Aim: To write a program to simulate the Go-Back-N (GBN) and Selective Repeat (SR) modes of the sliding window protocol used for reliable data transfer.

Objectives:
• Understand the concept of a sliding window for flow and error control.
• Implement Go-Back-N ARQ, in which the sender retransmits all frames from the lost frame onward.
• Implement Selective Repeat ARQ, in which only the specific lost/corrupted frame is retransmitted.
• Compare the efficiency and buffering requirements of the two schemes.

Theory:
The Sliding Window Protocol allows a sender to transmit multiple frames before requiring an acknowledgment, improving throughput compared to Stop-and-Wait, while still providing flow control by limiting the number of unacknowledged frames in transit (the window size).

In Go-Back-N ARQ, the receiver only accepts frames in strict sequence order and discards any out-of-order frame; if a frame is lost or corrupted, the sender must retransmit that frame and all subsequent frames already sent, even if some of them were received correctly.

In Selective Repeat ARQ, the receiver buffers out-of-order but correctly received frames and individually acknowledges each frame; only the specific frame(s) that were lost or corrupted need to be retransmitted, which is more efficient over high-error/high-latency links but requires more complex buffering at both ends.

Objectives:

• Implement a basic message exchange over a reliable, connection-oriented TCP stream.

Theory:
TCP (Transmission Control Protocol) provides a reliable, connection-oriented, byte-stream service between two endpoints. A TCP server creates a socket, binds it to an address/port, listens for incoming connections, and accepts a connection to obtain a dedicated socket for communication; a TCP client creates a socket and calls connect() to establish a three-way handshake with the server.

Once connected, both sides can use send()/recv() to exchange a stream of bytes; TCP guarantees in-order, error-checked delivery, which makes it well suited to applications like file transfer and remote procedure calls where data integrity matters.

A simple application-layer protocol can be layered on top of TCP by agreeing on a message format. For example, the calculator program can send an operator and two operands as a formatted string, and the server parses, computes, and returns the result.



Experiment No. 6

Aim: To write a program using UDP sockets to enable transfer of different file types (script, text, audio, and video, one file each) between two machines.

Objectives:
• Understand the connectionless, datagram-oriented nature of UDP.
• Implement chunked file transfer over UDP using sendto()/recvfrom().
• Handle files of different types (treated uniformly as binary data) over UDP.
• Observe the differences in reliability compared to the TCP-based transfer in Experiment 5.

Theory:
UDP (User Datagram Protocol) is a lightweight, connectionless transport protocol that sends independent datagrams without establishing a connection and without guaranteeing delivery, ordering, or duplicate protection. It has lower overhead than TCP, making it attractive for real-time or simple transfer applications, but the application must implement any reliability it needs.

For file transfer over UDP, a file is broken into fixed-size chunks; each chunk is sent as a separate datagram using sendto(), and the receiver reassembles the file from the datagrams received via recvfrom(). Because UDP does not guarantee order or delivery, sequence numbers and/or an end-of-file marker are commonly used at the application layer to detect missing or out-of-order chunks.

The same program logic applies regardless of file type (text, script, audio, or video) since all are transferred as raw binary bytes; only the source file changes between runs.

Experiment No. 7

Aim: To write programs using TCP sockets to implement (a) a Peer-to-Peer chat application and (b) a Multiuser (broadcast) chat application over a wired network.

Objectives:
• Implement bidirectional, real-time messaging between two peers using TCP.

Theory:
A Peer-to-Peer (P2P) chat over TCP requires simultaneous send and receive capability at both ends; this is commonly implemented by running the send and receive operations on separate threads so that a process can display incoming messages while still accepting user input to send.

A Multiuser chat is typically implemented using a central server that accepts multiple client connections, each handled on its own thread (or via an I/O multiplexing mechanism such as select()/poll()), and broadcasts any message received from one client to all other connected clients.

Because TCP is stream-oriented, application-level framing (e.g., newline-terminated messages) is used so that each message is correctly delimited when read by the receiving side.

Experiment No. 8

Aim: To write programs using UDP sockets to implement (a) a Peer-to-Peer chat application and (b) a Multiuser (broadcast) chat application over a wired network.

Objectives:
• Implement bidirectional chat between two peers using connectionless UDP datagrams.
• Implement a multiuser chat where a central server relays datagrams between multiple known clients.
• Compare the design and behavior of UDP-based chat with the TCP-based chat of Experiment 7.

Theory:
Unlike TCP, UDP has no notion of an established 'connection'. Each message is sent as an independent datagram using sendto(), addressed explicitly to the destination IP and port, and received using recvfrom(), which also reports the sender's address. A UDP P2P chat therefore uses a single socket at each peer that both sends to and receives from the other peer's known address.

A UDP-based multiuser chat requires the server to explicitly track the address of every client that has sent it a datagram (since there is no persistent connection to enumerate), typically using a registration message; the server then forwards (broadcasts) each received message to every other registered client address using sendto().

Because UDP does not guarantee delivery or ordering, a UDP chat application may occasionally lose or reorder messages, which is acceptable for a lab exercise but would need additional reliability mechanisms for production use.
